class LocalAI:
    def __init__(self):
        pass

    def delegate_bash(self, command: str) -> str:
        """
        Helper function to run bash commands and return their output.

        :param command: The command to be executed.
        :return: Output of the command execution.
        """
        import subprocess
        try:
            result = subprocess.run(command, shell=True, check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
            return result.stdout.decode().strip()
        except subprocess.CalledProcessError as e:
            print(f"Error executing command: {e}")
            return ""

    def delegate_git_commit(self, context: str = "") -> str:
        """
        Analyzes staged/unstaged changes and generates a commit message following conventional commits.

        :param context: Additional context for the commit.
        :return: Commit message.
        """
        status_output = self.delegate_bash("git status")
        diff_output = self.delegate_bash("git diff")

        # Example logic to generate commit message
        if "modified:" in status_output or "new file:" in status_output:
            commit_message = f"feat: {context}"
        else:
            commit_message = "chore: Update files"

        return commit_message

    def delegate_pr_description(self, branch: str, base: str = "main") -> str:
        """
        Analyzes commits and changes between branches to generate a PR title + description.

        :param branch: The target branch for the pull request.
        :param base: The base branch of the pull request (default is 'main').
        :return: PR description in markdown format.
        """
        log_output = self.delegate_bash(f"git log {base}..{branch}")
        diff_output = self.delegate_bash(f"git diff {base}...{branch}")

        # Example logic to generate PR title and description
        commits = [commit.strip() for commit in log_output.split('\n') if commit]
        changes = [change.strip() for change in diff_output.split('\n') if change]

        pr_title = f"Merge pull request from {base}..{branch}"
        pr_description = f"# Pull Request Description\n\n"
        pr_description += f"> **Title:** {pr_title}\n\n"
        pr_description += "## Summary:\n"
        for commit in commits:
            pr_description += f"- {commit}\n"

        pr_description += "\n## Changes:\n"
        for change in changes:
            pr_description += f"- {change}\n"

        return pr_description

    def delegate_git_summary(self, commits: int = 10) -> str:
        """
        Summarizes recent commits.

        :param commits: Number of recent commits to summarize.
        :return: Concise summary of the specified number of commits.
        """
        log_output = self.delegate_bash(f"git log -n {commits} --stat")
        return log_output
