**Improved Parse Action Function**

Here's the updated `parse_action()` function:
```python
import re

def parse_action(action_text):
    # Regular expression to match action format
    pattern = r"Action: (.*?)(\(([^)]+)\))?"
    
    match = re.match(pattern, action_text)
    
    if match:
        tool_name = match.group(1).strip()
        args = {}
        
        if match.group(2):
            arg_str = match.group(2)[1:-1].split(", ")
            for arg in arg_str:
                key, value = arg.split("=")
                args[key] = value.strip('"')
        
        return {"tool_name": tool_name, "args": args}
    
    # Fallback parsing for common patterns
    elif re.match(r"Action: (.*)", action_text):
        return {"tool_name": match.group(1).strip()}
    
    else:
        raise ValueError("Invalid action format")
```
