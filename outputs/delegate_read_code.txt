import logging

class LocalAI:
    def __init__(self):
        self.logger = logging.getLogger(__name__)
    
    def delegate_read(
        self,
        file_path: str,
        goal: str,
        max_output_chars: int = 2000
    ) -> str:
        """
        Read file via RTX and extract only relevant parts.
        
        Instead of reading full file (10k+ tokens), RTX:
        1. Reads the file
        2. Analyzes content based on goal
        3. Returns only relevant sections
        
        Args:
            file_path: Path to file
            goal: What to look for (e.g., "find login function")
            max_output_chars: Max chars in result
        
        Returns:
            Relevant file excerpts with context
        
        Example:
            # Before: Read entire 5000-line file (20k tokens)
            # After: Get only relevant 100 lines (400 tokens)
            
            result = delegate_read(
                "/c/ClaudeAgent/rtx_agent.py",
                "how does the ReAct loop work?"
            )
        """
        try:
            self.logger.info(f"[READ] Starting to read file: {file_path} with goal: {goal}")
            # Simulate reading and analyzing the file
            relevant_lines = []
            with open(file_path, 'r') as file:
                lines = file.readlines()
                for i, line in enumerate(lines):
                    if goal.lower() in line.lower():
                        relevant_lines.append(f"{i+1}: {line.strip()}")
                        # Stop after max_output_chars characters are collected
                        if sum(len(line) for line in relevant_lines) > max_output_chars:
                            break
            
            result = "\n".join(relevant_lines)
            self.logger.info(f"[READ] Found and returned: {result}")
            return result
        
        except Exception as e:
            self.logger.error(f"[READ] Error reading file: {e}", exc_info=True)
            return ""
