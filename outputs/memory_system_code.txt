import os
from typing import Dict, List
import logging
from qdrant_client import QdrantClient
from sentence_transformers import SentenceTransformer

class QdrantMemory:
    """
    In-memory Qdrant-based memory system for rtx_agent.py.
    
    Attributes:
        client (QdrantClient): In-memory Qdrant client.
        model (SentenceTransformer): All-MiniLM-L6-v2 model for local embeddings.
        collection_name (str): Name of the auto-created collection.
    """

    def __init__(self):
        """
        Initialize the memory system.
        
        Creates an in-memory Qdrant client, loads the all-MiniLM-L6-v2 model,
        and auto-creates a new collection named "memories".
        """
        self.client = QdrantClient()
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
        self.collection_name = 'memories'
        self.client.create_collection(self.collection_name)

    def add_memory(self, task: str, result: str, metadata: Dict) -> None:
        """
        Add a new memory to the system.
        
        Args:
            task (str): Task description.
            result (str): Result of the task.
            metadata (Dict): Additional metadata about the memory.
        
        Stores the task description, result, timestamp, and metadata in Qdrant.
        """
        vector = self.model.encode([task], convert_to_tensor=True)[0]
        self.client.indexing.add_documents([
            {
                'vector': vector,
                'data': {'task': task, 'result': result, 'metadata': metadata},
                'timestamp': int(os.time()),
            }
        ])

    def search_memories(self, query: str, limit: int = 5) -> List[Dict]:
        """
        Search for memories using semantic search.
        
        Args:
            query (str): Query to search for.
            limit (int): Maximum number of results to return. Defaults to 5.
        
        Returns:
            List[Dict]: Top-k relevant past experiences.
        """
        vector = self.model.encode([query], convert_to_tensor=True)[0]
        results = self.client.search(self.collection_name, vector, limit=limit)
        memories = []
        for result in results:
            memory = {
                'task': result.data['task'],
                'result': result.data['result'],
                'metadata': result.data['metadata']
            }
            memories.append(memory)
        return memories

if __name__ == '__main__':
    logging.basicConfig(level=logging.INFO)
    memory_system = QdrantMemory()
    # Example usage:
    memory_system.add_memory('Task 1', 'Result 1', {'foo': 'bar'})
    memory_system.add_memory('Task 2', 'Result 2', {'baz': 'qux'})
    query = 'Find the result of Task 1'
    results = memory_system.search_memories(query)
    for memory in results:
        logging.info(memory)

This code creates an in-memory Qdrant client and loads the all-MiniLM-L6-v2 model for local embeddings. The `add_memory` method adds a new memory to the system, storing the task description, result, timestamp, and metadata in Qdrant. The `search_memories` method performs semantic search using the query's embedding and returns the top-k relevant past experiences.
